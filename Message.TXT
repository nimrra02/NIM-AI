  normal:        { latency:180, cpu:54,  db:110, traffic:'baseline', cls:'' },

  investigating: { latency:740, cpu:87,  db:380, traffic:'+63%',     cls:'critical' },

  fixing:        { latency:740, cpu:87,  db:380, traffic:'+63%',     cls:'critical' },

const RESOLVED_TEXT = 'Infrastructure health has returned to normal.';

let phase = 'normal';

function goto(view){
  document.querySelectorAll('.view').forEach(v=>v.classList.remove('show'));
  document.getElementById('view-'+view).classList.add('show');
  document.querySelectorAll('.nav-item').forEach(n=>n.classList.remove('active-nav'));
  document.querySelector('.nav-item[data-view="'+view+'"]').classList.add('active-nav');
  renderAll();
}

function stagesFor(p){
  switch(p){
    case 'normal':        return { active:null, done:[] };
    case 'critical':      return { active:'detect', done:[] };
    case 'investigating': return { active:'investigate', done:['detect'] };
    case 'recommended':   return { active:'approve', done:['detect','investigate','recommend'] };
    case 'fixing':        return { active:'fix', done:['detect','investigate','recommend','approve'] };
    case 'resolved':      return { active:null, done:['detect','investigate','recommend','approve','fix','verify'] };
  }
}

function triggerIncident(){
  if(phase !== 'normal') return;
  phase = 'critical';
  renderAll();
}

function agentAction(){
  if(phase === 'critical'){
    phase = 'investigating';
    renderAll();
    setTimeout(()=>{
      phase = 'recommended';
      renderAll();
    }, 2000);
  } else if(phase === 'recommended'){
    phase = 'fixing';
    renderAll();
    setTimeout(()=>{
      phase = 'resolved';
      renderAll();
    }, 1400);
  }
}

function resetDemo(){
  phase = 'normal';
  renderAll();
  goto('dashboard');
}

  /* ---- Dashboard ---- */
